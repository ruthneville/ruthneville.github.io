\section{Profile}
% ===============================================================

{\small
Population geographer and computational social scientist specialising in
international student mobility, migration systems, demographic forecasting,
and population change under disruption. My research examines how educational
and migration pathways are shaped by policy, demographic change, institutional
relationships, and shocks, and how these processes redistribute people, skills,
and opportunities across places. My emerging research agenda examines
international student mobility as both an educational trajectory and an
upstream component of high-skilled migration and human-capital formation.
}

\skillline{Methods}{
Spatial interaction modelling; longitudinal and spatial analysis;
difference-in-differences and causal inference; demographic forecasting;
machine learning; network analysis; clustering; data linkage;
data visualisation; reproducible research workflows
}

\skillline{Programming}{
R (advanced: \texttt{tidyverse}, \texttt{sf}, \texttt{ggplot2},
\texttt{tmap}, R Markdown, Quarto);
Python (intermediate: \texttt{pandas}, \texttt{scikit-learn},
\texttt{matplotlib});
QGIS; \LaTeX; Git/GitHub
}
